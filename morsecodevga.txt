`default_nettype none

module tt_um_vga_example (
    input wire [7:0] ui_in,
    output wire [7:0] uo_out,
    input wire [7:0] uio_in,
    output wire [7:0] uio_out, uio_oe,
    input wire ena, clk, rst_n
);

    reg [9:0] x, y;
    always @(posedge clk) begin
        if (!rst_n) begin x <= 0; y <= 0; end
        else if (x == 799) begin
            x <= 0;
            if (y == 524) y <= 0; else y <= y + 1'b1;
        end else x <= x + 1'b1;
    end
    wire hs = !(x >= 656 && x < 752);
    wire vs = !(y >= 490 && y < 492);
    wire active = x < 640 && y < 480;

    reg [7:0] sync1, sync2, previous;
    wire [7:0] press = sync2 & ~previous;
    always @(posedge clk) begin
        if (!rst_n) begin sync1 <= 0; sync2 <= 0; previous <= 0; end
        else begin sync1 <= ui_in; sync2 <= sync1; previous <= sync2; end
    end

    reg [4:0] pattern;
    reg [2:0] length;
    reg overflow;
    reg [7:0] text_mem [0:255];
    reg [7:0] cursor;
    integer i;

    function [7:0] decode;
        input [2:0] n;
        input [4:0] p;
        begin
            decode = "?";
            case ({n,p})
                8'b010_00001: decode = "A";
                8'b100_01000: decode = "B";
                8'b100_01010: decode = "C";
                8'b011_00100: decode = "D";
                8'b001_00000: decode = "E";
                8'b100_00010: decode = "F";
                8'b011_00110: decode = "G";
                8'b100_00000: decode = "H";
                8'b010_00000: decode = "I";
                8'b100_00111: decode = "J";
                8'b011_00101: decode = "K";
                8'b100_00100: decode = "L";
                8'b010_00011: decode = "M";
                8'b010_00010: decode = "N";
                8'b011_00111: decode = "O";
                8'b100_00110: decode = "P";
                8'b100_01101: decode = "Q";
                8'b011_00010: decode = "R";
                8'b011_00000: decode = "S";
                8'b001_00001: decode = "T";
                8'b011_00001: decode = "U";
                8'b100_00001: decode = "V";
                8'b011_00011: decode = "W";
                8'b100_01001: decode = "X";
                8'b100_01011: decode = "Y";
                8'b100_01100: decode = "Z";
                8'b101_11111: decode = "0";
                8'b101_01111: decode = "1";
                8'b101_00111: decode = "2";
                8'b101_00011: decode = "3";
                8'b101_00001: decode = "4";
                8'b101_00000: decode = "5";
                8'b101_10000: decode = "6";
                8'b101_11000: decode = "7";
                8'b101_11100: decode = "8";
                8'b101_11110: decode = "9";
                default: decode = "?";
            endcase
        end
    endfunction

    always @(posedge clk) begin
        if (!rst_n) begin
            pattern <= 0; length <= 0; overflow <= 0; cursor <= 0;

            for (i=0; i<256; i=i+1) text_mem[i] = " ";
        end else if (press[7]) begin

            if (overflow) overflow <= 0;
            else if (length != 0) begin
                pattern <= {1'b0, pattern[4:1]};
                length <= length - 1'b1;
            end
        end else if (press[2]) begin
            if (length != 0 || overflow) begin
                text_mem[cursor] <= overflow ? "?" : decode(length, pattern);
                cursor <= cursor + 1'b1;
            end
            pattern <= 0; length <= 0; overflow <= 0;
        end else if (press[3]) begin
            text_mem[cursor] <= " ";
            cursor <= cursor + 1'b1;
            pattern <= 0; length <= 0; overflow <= 0;
        end else if (press[0] || press[1]) begin
            if (length < 5) begin
                pattern <= {pattern[3:0], !press[0]};
                length <= length + 1'b1;
            end else overflow <= 1;
        end
    end

    function [34:0] glyph;
        input [7:0] ch;
        begin
            case (ch)
                "A": glyph = 35'b01110_10001_10001_11111_10001_10001_10001;
                "B": glyph = 35'b11110_10001_10001_11110_10001_10001_11110;
                "C": glyph = 35'b01111_10000_10000_10000_10000_10000_01111;
                "D": glyph = 35'b11110_10001_10001_10001_10001_10001_11110;
                "E": glyph = 35'b11111_10000_10000_11110_10000_10000_11111;
                "F": glyph = 35'b11111_10000_10000_11110_10000_10000_10000;
                "G": glyph = 35'b01111_10000_10000_10111_10001_10001_01111;
                "H": glyph = 35'b10001_10001_10001_11111_10001_10001_10001;
                "I": glyph = 35'b01110_00100_00100_00100_00100_00100_01110;
                "J": glyph = 35'b00111_00010_00010_00010_10010_10010_01100;
                "K": glyph = 35'b10001_10010_10100_11000_10100_10010_10001;
                "L": glyph = 35'b10000_10000_10000_10000_10000_10000_11111;
                "M": glyph = 35'b10001_11011_10101_10101_10001_10001_10001;
                "N": glyph = 35'b10001_11001_10101_10011_10001_10001_10001;
                "O": glyph = 35'b01110_10001_10001_10001_10001_10001_01110;
                "P": glyph = 35'b11110_10001_10001_11110_10000_10000_10000;
                "Q": glyph = 35'b01110_10001_10001_10001_10101_10010_01101;
                "R": glyph = 35'b11110_10001_10001_11110_10100_10010_10001;
                "S": glyph = 35'b01111_10000_10000_01110_00001_00001_11110;
                "T": glyph = 35'b11111_00100_00100_00100_00100_00100_00100;
                "U": glyph = 35'b10001_10001_10001_10001_10001_10001_01110;
                "V": glyph = 35'b10001_10001_10001_10001_10001_01010_00100;
                "W": glyph = 35'b10001_10001_10001_10101_10101_10101_01010;
                "X": glyph = 35'b10001_10001_01010_00100_01010_10001_10001;
                "Y": glyph = 35'b10001_10001_01010_00100_00100_00100_00100;
                "Z": glyph = 35'b11111_00001_00010_00100_01000_10000_11111;
                "0": glyph = 35'b01110_10001_10011_10101_11001_10001_01110;
                "1": glyph = 35'b00100_01100_00100_00100_00100_00100_01110;
                "2": glyph = 35'b01110_10001_00001_00010_00100_01000_11111;
                "3": glyph = 35'b11110_00001_00001_01110_00001_00001_11110;
                "4": glyph = 35'b00010_00110_01010_10010_11111_00010_00010;
                "5": glyph = 35'b11111_10000_10000_11110_00001_00001_11110;
                "6": glyph = 35'b01110_10000_10000_11110_10001_10001_01110;
                "7": glyph = 35'b11111_00001_00010_00100_01000_01000_01000;
                "8": glyph = 35'b01110_10001_10001_01110_10001_10001_01110;
                "9": glyph = 35'b01110_10001_10001_01111_00001_00001_01110;
                "?": glyph = 35'b01110_10001_00001_00010_00100_00000_00100;
                default: glyph = 0;
            endcase
        end
    endfunction

    wire [9:0] tx = x - 10'd64;
    wire [9:0] ty = y - 10'd64;
    wire in_text = x >= 64 && x < 576 && y >= 64 && y < 320;
    wire [7:0] address = {ty[7:5], tx[8:4]};
    wire [2:0] gx = tx[3:1];
    wire [3:0] gy = ty[4:1];
    reg [34:0] bitmap;
    reg ink;
    reg [7:0] preview_ch;
    reg [2:0] slot;
    always @* begin
        bitmap = glyph(text_mem[address]);
        ink = 0;
        preview_ch = " ";
        slot = tx[6:4];
        if (in_text && gx < 5 && gy < 7)
            ink = bitmap[34 - gy*5 - gx];

        if (x >= 64 && x < 144 && y >= 352 && y < 368) begin
            if (slot < length)
                preview_ch = pattern[length-1-slot] ? "-" : ".";
            bitmap = glyph(preview_ch);
            if (gx < 5 && y[3:1] < 7)
                ink = bitmap[34 - y[3:1]*5 - gx];
        end
    end
    wire caret = in_text && address == cursor && ty[4:0] == 20 && tx[3:0] < 12;
    wire [1:0] red = active && ink ? 2'b11 : 2'b00;
    wire [1:0] green = active && (ink || caret) ? 2'b11 : 2'b00;
    wire [1:0] blue = active && ink ? 2'b11 : 2'b00;
    assign uo_out = {hs, blue[0], green[0], red[0], vs, blue[1], green[1], red[1]};
    assign uio_out = 0;
    assign uio_oe = 0;
    wire unused_ok = &{ena, uio_in, ui_in[6:4], 1'b0};
endmodule
`default_nettype wire
